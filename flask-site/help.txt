Простая выборка:

cursor.execute("SELECT * FROM users")
data = cursor.fetchall()
for row in data:
    print(row)
==
Если нужно выбрать только определённые столбцы:

cursor.execute("SELECT id, username FROM users")
==
Выборка с фильтром:

cursor.execute("SELECT * FROM users WHERE id = %s", (1,))
==
Добавление данных

cursor.execute("INSERT INTO users (username, age) VALUES (%s, %s)", ("Alice", 25))
conn.commit()
==
Обновление данных

cursor.execute("UPDATE users SET age = %s WHERE username = %s", (26, "Alice"))
conn.commit()
==
Удаление данных

cursor.execute("DELETE FROM users WHERE username = %s", ("Alice",))
conn.commit()
==
Закрытие соединения

conn.close()
